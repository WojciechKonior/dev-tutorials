Array(const Array& rhs) { 
    printf("Copy constructor\n"); 
    data = new int[10]; 
    for(int i = 0; i<10; i++) {
        data[i] = rhs.data[i]; 
    } 
}
Array& operator=(const Array& rhs) { 
    printf("copy-assignment operator\n"); 
    if(&rhs == this) return *this;
    if(data != nullptr) delete[] data;
    data = new int[10];
    for(int i = 0; i<10; i++){
        data[i] = rhs.data[i];
    }
    return *this;
}
Array(Array&& rhs){
    printf("move constructor\n");
    if(data != nullptr) delete[] data;
    data = rhs.data;
    rhs.data = nullptr;
}
Array& operator=(Array&& rhs){
    printf("move assignment constructor\n");
    if(&rhs == this) return *this;
    if(data != nullptr) delete[] data;
    data = rhs.data;
    rhs.data = nullptr;
    return *this;
}
